CREATE TABLE public.tax (
	tax_id int4 NOT NULL GENERATED BY DEFAULT AS IDENTITY,
	"year" int4 NOT NULL,
	"month" int4 NOT NULL,
	tax_paid numeric(12, 2) NOT NULL DEFAULT 0,
	taxable_income numeric(12, 2) NULL DEFAULT 0,
	status varchar(20) NOT NULL DEFAULT 'paid'::character varying,
	payment_date date NULL,
	payment_mode varchar(50) NULL DEFAULT 'Online'::character varying,
	notes text NULL,
	created_at timestamptz NOT NULL DEFAULT now(),
	updated_at timestamptz NOT NULL DEFAULT now(),
	is_deleted bool NOT NULL DEFAULT false,
	CONSTRAINT tax_entries_month_check CHECK ((((month >= 1) AND (month <= 12)))),
	CONSTRAINT tax_entries_status_check CHECK ((((status)::text = ANY ((ARRAY['paid'::character varying, 'pending'::character varying, 'overdue'::character varying])::text[])))),
	CONSTRAINT tax_entries_tax_paid_check CHECK (((tax_paid >= (0)::numeric))),
	CONSTRAINT tax_entries_taxable_income_check CHECK (((taxable_income >= (0)::numeric))),
	CONSTRAINT tax_entries_year_check CHECK ((((year >= 2021) AND (year <= 2030)))),
	CONSTRAINT tax_pkey PRIMARY KEY (tax_id),
	CONSTRAINT ux_tax_year_month UNIQUE ("year","month")
);
CREATE INDEX idx_tax_active ON public.tax (is_deleted);
CREATE INDEX idx_tax_month ON public.tax ("month");
CREATE INDEX idx_tax_status ON public.tax (status);
CREATE INDEX idx_tax_year ON public.tax ("year");
CREATE INDEX idx_tax_year_month ON public.tax ("year","month");
