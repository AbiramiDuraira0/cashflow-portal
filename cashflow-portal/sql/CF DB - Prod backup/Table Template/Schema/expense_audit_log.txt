CREATE TABLE public.expense_audit_log (
	id serial4 NOT NULL,
	"timestamp" timestamptz NOT NULL DEFAULT now(),
	operation varchar(20) NOT NULL,
	table_name varchar(50) NOT NULL,
	expense_id int4 NULL,
	"year" int4 NOT NULL,
	"month" varchar(20) NULL,
	request_data jsonb NULL,
	response_data jsonb NULL,
	before_data jsonb NULL,
	user_agent text NULL,
	status varchar(10) NOT NULL,
	error_message text NULL,
	stack_trace text NULL,
	created_at timestamptz NOT NULL DEFAULT now(),
	CONSTRAINT expense_audit_log_operation_check CHECK ((((operation)::text = ANY ((ARRAY['INSERT'::character varying, 'UPDATE'::character varying, 'DELETE'::character varying, 'SOFT_DELETE'::character varying, 'LOAD'::character varying, 'YEAR_MOVE'::character varying])::text[])))),
	CONSTRAINT expense_audit_log_pkey PRIMARY KEY (id),
	CONSTRAINT expense_audit_log_status_check CHECK ((((status)::text = ANY ((ARRAY['SUCCESS'::character varying, 'ERROR'::character varying])::text[]))))
);
CREATE INDEX idx_expense_audit_expense_id ON public.expense_audit_log (expense_id);
CREATE INDEX idx_expense_audit_operation ON public.expense_audit_log (operation);
CREATE INDEX idx_expense_audit_table_name ON public.expense_audit_log (table_name);
CREATE INDEX idx_expense_audit_timestamp ON public.expense_audit_log ("timestamp");
CREATE INDEX idx_expense_audit_year ON public.expense_audit_log ("year");
