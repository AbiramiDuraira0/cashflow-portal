CREATE TABLE public.lifeline (
	lifeline_id int4 NOT NULL GENERATED BY DEFAULT AS IDENTITY,
	entry_date date NOT NULL,
	amount_inr numeric(12, 2) NOT NULL,
	notes text NULL,
	is_delete bool NOT NULL DEFAULT false,
	created_at timestamptz NOT NULL DEFAULT now(),
	updated_at timestamptz NOT NULL DEFAULT now(),
	CONSTRAINT lifeline_amount_inr_check CHECK (((amount_inr >= (0)::numeric))),
	CONSTRAINT lifeline_pkey PRIMARY KEY (lifeline_id)
);
CREATE INDEX idx_lifeline_active ON public.lifeline (is_delete);
CREATE INDEX idx_lifeline_entry_date ON public.lifeline (entry_date DESC);
