CREATE TABLE public.expense_2023 (
	expense_id int4 NOT NULL GENERATED BY DEFAULT AS IDENTITY,
	"month" varchar(20) NOT NULL,
	"year" int4 NOT NULL DEFAULT 2023,
	category_id int4 NOT NULL,
	category_name varchar(50) NOT NULL,
	category_icon varchar(10) NULL,
	sub_category varchar(100) NULL,
	subcategory_icon varchar(10) NULL,
	amount_inr numeric(12, 2) NOT NULL,
	notes text NULL,
	is_delete bool NOT NULL DEFAULT false,
	created_at timestamptz NOT NULL DEFAULT now(),
	updated_at timestamptz NOT NULL DEFAULT now(),
	CONSTRAINT chk_expense_2023_year CHECK (((year = 2023))),
	CONSTRAINT expense_2023_amount_inr_check CHECK (((amount_inr >= (0)::numeric))),
	CONSTRAINT expense_2023_pkey PRIMARY KEY (expense_id)
);
CREATE INDEX idx_expense_2023_category ON public.expense_2023 (category_name);
CREATE INDEX idx_expense_2023_is_delete ON public.expense_2023 (is_delete);
CREATE INDEX idx_expense_2023_year_month ON public.expense_2023 ("year","month");
