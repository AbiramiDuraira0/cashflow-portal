CREATE TABLE public.debts (
	debt_id int4 NOT NULL GENERATED BY DEFAULT AS IDENTITY,
	debt_type varchar(20) NOT NULL,
	loan_name varchar(100) NOT NULL,
	bank_or_person varchar(100) NOT NULL,
	principal_amount numeric(12, 2) NOT NULL,
	interest_rate numeric(5, 2) NULL,
	emi_amount numeric(12, 2) NULL,
	emi_start_date date NULL,
	emi_end_date date NULL,
	outstanding_amount numeric(12, 2) NOT NULL,
	amount_paid numeric(12, 2) NOT NULL DEFAULT 0,
	status varchar(20) NOT NULL DEFAULT 'open'::character varying,
	notes text NULL,
	is_delete bool NOT NULL DEFAULT false,
	created_at timestamptz NOT NULL DEFAULT now(),
	updated_at timestamptz NOT NULL DEFAULT now(),
	CONSTRAINT debts_amount_paid_check CHECK ((amount_paid >= (0)::numeric)),
	CONSTRAINT debts_debt_type_check CHECK (((debt_type)::text = ANY ((ARRAY['debt'::character varying, 'receivable'::character varying])::text[]))),
	CONSTRAINT debts_emi_amount_check CHECK ((emi_amount >= (0)::numeric)),
	CONSTRAINT debts_outstanding_amount_check CHECK ((outstanding_amount >= (0)::numeric)),
	CONSTRAINT debts_pkey PRIMARY KEY (debt_id),
	CONSTRAINT debts_principal_amount_check CHECK ((principal_amount >= (0)::numeric)),
	CONSTRAINT debts_status_check CHECK (((status)::text = ANY ((ARRAY['open'::character varying, 'closed'::character varying])::text[])))
);
CREATE INDEX idx_debts_bank_person ON public.debts USING btree (bank_or_person);
CREATE INDEX idx_debts_is_delete ON public.debts USING btree (is_delete);
CREATE INDEX idx_debts_status ON public.debts USING btree (status);
CREATE INDEX idx_debts_type ON public.debts USING btree (debt_type);
